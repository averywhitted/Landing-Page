// Served as JavaScript at /admin/app.js (stored as .txt so it bundles into the service as text).
// Booking admin (served at book.averywhitted.com/admin behind Cloudflare Access).
(() => {
  "use strict";
  const TZ = "America/New_York";
  const $main = document.getElementById("main");
  const $tabs = document.getElementById("tabs");
  const $dlg = document.getElementById("dlg");
  let data = null;
  let tab = (location.hash || "#upcoming").slice(1);

  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const money = (c) => (c % 100 ? `$${(c / 100).toFixed(2)}` : `$${c / 100}`);
  const fmt = (iso, o) => new Intl.DateTimeFormat("en-US", { timeZone: TZ, ...o }).format(Date.parse(iso));
  const dayLabel = (iso) => fmt(iso, { weekday: "long", month: "long", day: "numeric" });
  const clock = (iso) => fmt(iso, { hour: "numeric", minute: "2-digit" });
  const minutes = (b) => Math.round((Date.parse(b.end) - Date.parse(b.start)) / 60000);

  async function api(method, path, body) {
    const r = await fetch(path, {
      method,
      headers: { "X-Admin": "1", ...(body ? { "Content-Type": "application/json" } : {}) },
      body: body ? JSON.stringify(body) : undefined,
      cache: "no-store",
      credentials: "same-origin",
    });
    const out = await r.json().catch(() => ({}));
    if (!r.ok) throw new Error(out.error || `Request failed (${r.status})`);
    return out;
  }

  function toast(text, isErr = false) {
    const t = document.createElement("div");
    t.className = `toast${isErr ? " err" : ""}`;
    t.setAttribute("role", isErr ? "alert" : "status");
    t.textContent = text;
    document.body.appendChild(t);
    setTimeout(() => t.remove(), 4000);
  }

  async function load() {
    try {
      data = await api("GET", "/api/admin/overview");
      render();
    } catch (err) {
      $main.innerHTML = `<p class="empty">${esc(err.message)}</p>`;
    }
  }

  const refundsDue = () => data.packages.filter((p) => p.cancelled && p.refundDueCents && !p.refunded);
  const sessionRefunds = () => data.bookings.filter((b) => b.refundOwed);
  function problemsCount() {
    return data.problems.failedEmails.length + data.problems.stuck.length + refundsDue().length
      + sessionRefunds().length + data.problems.leftOnCalendar.length;
  }

  function renderTabs() {
    // A group session counts once, however many students it has.
    const upcoming = new Set(data.bookings.filter((b) => b.status === "confirmed" && !b.past).map((b) => b.groupId || b.id)).size;
    const tabs = [
      ["upcoming", "Upcoming", upcoming],
      ["recent", "Recent", null],
      ["bundles", "Bundles", data.packages.length],
      ["attention", "Needs attention", problemsCount()],
      ["settings", "Settings", null],
    ];
    $tabs.innerHTML = tabs.map(([id, label, n]) => `<button class="tab" role="tab" aria-selected="${id === tab}" data-tab="${id}">${label}${n !== null ? `<span class="count${id === "attention" && n ? " alert" : ""}">${n}</span>` : ""}</button>`).join("");
  }

  // How a booking is paid for, as shown on its card.
  const payLabel = (b) => b.paid === "bundle" ? "Bundle credit"
    : b.paid ? money(b.paid) + (b.promoCode ? ` (${b.promoCode})` : "")
    : b.dueCents ? `${money(b.dueCents)} due${b.payBy ? ` by ${fmt(b.payBy, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" })}` : ""}`
    : "Free";

  function bookingCard(b) {
    const paid = payLabel(b);
    const chips = [
      `<span class="chip${b.dueCents ? " bad" : ""}">${esc(paid)}</span>`,
      b.byAvery ? '<span class="chip">Booked by you</span>' : "",
      b.status === "cancelled" ? `<span class="chip bad">${b.unpaidReleased ? "Released (unpaid)" : `Cancelled${b.cancelReason === "avery_cancelled" ? " by you" : b.cancelReason === "client_cancelled" ? " by client" : ""}`}</span>` : "",
      b.refunded ? '<span class="chip ok">Refunded</span>' : "",
      b.refundOwed === "auto" ? '<span class="chip bad">Refund retrying</span>' : "",
      b.refundOwed === "choice" ? '<span class="chip">Not refunded</span>' : "",
      b.status === "confirmed" && !b.past ? `<span class="chip ${b.inCalendar ? "ok" : "bad"}">${b.inCalendar ? "In calendar" : "Not in calendar"}</span>` : "",
      b.status === "confirmed" && !b.past ? `<span class="chip ${b.emailed ? "ok" : "bad"}">${b.emailed ? "Emailed" : "Not emailed"}</span>` : "",
    ].join("");
    const intake = b.intake ? Object.entries({ Goal: b.intake.goal, Material: b.intake.material, Link: b.intake.link, Notes: b.intake.notes })
      .filter(([, v]) => v).map(([k, v]) => `<dt>${k}</dt><dd>${k === "Link" ? `<a href="${esc(v)}" target="_blank" rel="noopener">${esc(v)}</a>` : esc(v)}</dd>`).join("") : "";
    const canCancel = b.status === "confirmed" && !b.past;
    return `<div class="card${b.status === "cancelled" || b.past ? " muted" : ""}">
      <div class="time">${esc(clock(b.start))}<small>${minutes(b)} min</small></div>
      <div class="who">
        <strong>${esc(b.name)}</strong>${b.pronouns ? ` <span class="sub" style="display:inline">(${esc(b.pronouns)})</span>` : ""}
        <div class="sub"><a href="mailto:${esc(b.email)}">${esc(b.email)}</a> &middot; ${esc(b.service)}</div>
        <div class="chips">${chips}</div>
        ${intake ? `<details class="intake"><summary>Intake answers</summary><dl>${intake}</dl></details>` : ""}
      </div>
      <div class="actions">
        ${b.zoomUrl && canCancel ? `<a class="btn primary" href="${esc(b.zoomUrl)}" target="_blank" rel="noopener">Zoom</a>` : ""}
        ${b.stripeUrl ? `<a class="btn" href="${esc(b.stripeUrl)}" target="_blank" rel="noopener">Stripe</a>` : ""}
        ${canCancel ? `<button class="btn" data-move="${esc(b.id)}">Move</button>` : ""}
        ${canCancel ? `<button class="btn" data-resend="${esc(b.id)}">Resend invite</button>` : ""}
        ${canCancel ? `<button class="btn danger" data-cancel="${esc(b.id)}">Cancel</button>` : ""}
        ${b.refundOwed === "choice" ? `<button class="btn" data-refund="${esc(b.id)}">Refund ${money(b.paid)}</button>` : ""}
      </div>
    </div>`;
  }

  function byDay(list) {
    // Students in the same group session become one card.
    const items = [];
    const groups = new Map();
    for (const b of list) {
      if (!b.groupId) { items.push({ start: b.start, one: b }); continue; }
      if (!groups.has(b.groupId)) { const g = { start: b.start, seats: [] }; groups.set(b.groupId, g); items.push(g); }
      groups.get(b.groupId).seats.push(b);
    }
    let out = "";
    let last = "";
    for (const it of items) {
      const d = dayLabel(it.start);
      if (d !== last) { out += `<p class="day">${esc(d)}</p>`; last = d; }
      out += it.one ? bookingCard(it.one) : groupCard(it.seats);
    }
    return out;
  }

  function groupCard(seats) {
    const first = seats[0];
    const active = seats.filter((s) => s.status === "confirmed");
    const live = active.length && !first.past;
    const rows = seats.map((s) => `<div class="seat">
        <div class="who-line"><strong>${esc(s.name)}</strong>${s.pronouns ? ` (${esc(s.pronouns)})` : ""}<small><a href="mailto:${esc(s.email)}">${esc(s.email)}</a></small></div>
        <span class="chip${s.dueCents ? " bad" : s.paid && s.paid !== "bundle" ? " ok" : ""}">${esc(payLabel(s))}</span>
        ${s.status === "cancelled" ? `<span class="chip bad">${s.unpaidReleased ? "Released (unpaid)" : "Cancelled"}</span>` : ""}
        ${s.refunded ? '<span class="chip ok">Refunded</span>' : ""}
        <span class="seat-actions">
          ${s.refundOwed === "choice" ? `<button class="btn sm" data-refund="${esc(s.id)}">Refund ${money(s.paid)}</button>` : ""}
          ${live && s.status === "confirmed" ? `<button class="btn sm" data-resend="${esc(s.id)}">Resend</button><button class="btn sm danger" data-cancel="${esc(s.id)}">Remove</button>` : ""}
        </span>
      </div>`).join("");
    return `<div class="card${live ? "" : " muted"}">
      <div class="time">${esc(clock(first.start))}<small>${minutes(first)} min</small></div>
      <div class="who">
        <strong>Group session</strong> <span class="sub" style="display:inline">&middot; ${active.length} of ${seats.length} student${seats.length === 1 ? "" : "s"}</span>
        <div class="chips"><span class="chip ${first.inCalendar ? "ok" : "bad"}">${first.inCalendar ? "In calendar" : "Not in calendar"}</span></div>
        <div style="margin-top:8px">${rows}</div>
      </div>
      <div class="actions">
        ${first.zoomUrl && live ? `<a class="btn primary" href="${esc(first.zoomUrl)}" target="_blank" rel="noopener">Zoom</a>` : ""}
        ${live ? `<button class="btn" data-move-group="${esc(first.groupId)}" data-start="${esc(first.start)}">Move</button>` : ""}
        ${live ? `<button class="btn danger" data-cancel-group="${esc(first.groupId)}">Cancel session</button>` : ""}
      </div>
    </div>`;
  }

  function renderUpcoming() {
    const list = data.bookings.filter((b) => b.status === "confirmed" && !b.past);
    return list.length ? byDay(list) : '<p class="empty">No upcoming sessions.</p>';
  }

  function renderRecent() {
    const list = data.bookings.filter((b) => b.past || b.status === "cancelled").reverse();
    return list.length ? byDay(list) : '<p class="empty">Nothing in the last 30 days.</p>';
  }

  function renderBundles() {
    if (!data.packages.length) return '<p class="empty">No active bundles.</p>';
    return data.packages.map((p) => {
      const dots = Array.from({ length: p.creditsTotal }, (_, i) => `<span class="dot${i < p.creditsUsed ? " used" : ""}"></span>`).join("");
      if (p.cancelled) {
        return `<div class="card muted">
        <div class="time">0<small>cancelled</small></div>
        <div class="who"><strong>${esc(p.name)}</strong>
          <div class="sub"><a href="mailto:${esc(p.email)}">${esc(p.email)}</a> &middot; ${esc(p.bundle)} &middot; ${money(p.amountCents)}</div>
          <div class="chips">${p.refunded ? '<span class="chip ok">Refunded</span>' : p.refundDueCents ? `<span class="chip bad">Refund due ${money(p.refundDueCents)}</span>` : '<span class="chip">No refund due</span>'}</div>
        </div>
        <div class="actions">${p.stripeUrl && p.refundDueCents && !p.refunded ? `<a class="btn primary" href="${esc(p.stripeUrl)}" target="_blank" rel="noopener">Refund in Stripe</a>` : ""}</div>
      </div>`;
      }
      return `<div class="card${p.expired ? " muted" : ""}">
        <div class="time">${p.remaining}<small>of ${p.creditsTotal} left</small></div>
        <div class="who">
          <strong>${esc(p.name)}</strong>
          <div class="sub"><a href="mailto:${esc(p.email)}">${esc(p.email)}</a> &middot; ${esc(p.bundle)} &middot; ${money(p.amountCents)}${p.promoCode ? ` (${esc(p.promoCode)})` : ""}</div>
          <div class="meter" aria-hidden="true">${dots}</div>
          <div class="chips"><span class="chip ${p.expired ? "bad" : ""}">${p.expired ? "Expired" : "Use by"} ${esc(p.expiresAt ? fmt(p.expiresAt, { month: "short", day: "numeric", year: "numeric" }) : "")}</span></div>
        </div>
        <div class="actions">
          <button class="btn" data-credit="${esc(p.id)}" data-delta="1">+1 session</button>
          <button class="btn" data-credit="${esc(p.id)}" data-delta="-1"${p.remaining ? "" : " disabled"}>&minus;1 session</button>
          <button class="btn" data-extend="${esc(p.id)}">Extend</button>
        </div>
      </div>`;
    }).join("");
  }

  function renderAttention() {
    const { failedEmails, stuck, activeHolds } = data.problems;
    const parts = [];
    const owed = sessionRefunds();
    if (owed.length) {
      parts.push(`<div class="panel"><p class="day" style="margin-top:0">Session refunds</p>${owed.map((b) =>
        `<p>${esc(b.name)}, ${esc(dayLabel(b.start))}: <strong>${money(b.paid)}</strong> &middot; ${b.refundOwed === "auto"
          ? `automatic refund hasn't gone through yet${b.refundError ? ` (Stripe said: ${esc(b.refundError)})` : ""}. It keeps retrying.`
          : "you cancelled this one. Refund it if they'd like a refund, or leave it if they're rebooking."}
          <br><button class="btn" data-refund="${esc(b.id)}" style="margin-top:6px">Refund now</button>${b.stripeUrl ? ` <a class="btn" href="${esc(b.stripeUrl)}" target="_blank" rel="noopener" style="margin-top:6px">Stripe</a>` : ""}</p>`).join("")}</div>`);
    }
    const left = data.problems.leftOnCalendar;
    if (left.length) {
      parts.push(`<div class="panel"><p class="day" style="margin-top:0">Cancelled but still on your calendar</p>${left.map((s) =>
        `<p>${esc(s.name)}, ${esc(dayLabel(s.start))} at ${esc(clock(s.start))}. It isn't happening. Removal keeps retrying; you can also delete it from your Coaching calendar by hand.</p>`).join("")}</div>`);
    }
    const due = refundsDue();
    if (due.length) {
      parts.push(`<div class="panel"><p class="day" style="margin-top:0">Bundle refunds to issue</p>${due.map((p) =>
        `<p>${esc(p.name)}: <strong>${money(p.refundDueCents)}</strong> ${p.stripeUrl ? `&middot; <a href="${esc(p.stripeUrl)}" target="_blank" rel="noopener">Refund in Stripe</a>` : ""}<br><span style="font-size:.8125rem;color:var(--faint)">Marked done automatically once Stripe records the refund.</span></p>`).join("")}</div>`);
    }
    if (stuck.length) {
      parts.push(`<div class="panel"><p class="day" style="margin-top:0">Bookings missing a step</p>${stuck.map((s) =>
        `<p>${esc(s.name)}, ${esc(dayLabel(s.start))} at ${esc(clock(s.start))}: ${[!s.inCalendar ? "not in your Coaching calendar" : "", !s.emailed ? "confirmation email not sent" : ""].filter(Boolean).join(" and ")}. It retries every few minutes.</p>`).join("")}</div>`);
    }
    if (failedEmails.length) {
      parts.push(`<div class="panel"><p class="day" style="margin-top:0">Emails that failed (last 14 days)</p>${failedEmails.map((e) =>
        `<p><strong>${esc(e.kind.replace(/_/g, " "))}</strong> &middot; ${esc(fmt(e.at, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" }))}<br><span style="font-family:var(--mono);font-size:.8125rem;color:var(--faint)">${esc(e.error || "")}</span></p>`).join("")}</div>`);
    }
    parts.push(`<div class="panel"><p class="day" style="margin-top:0">Checkouts in progress</p><p>${activeHolds} time${activeHolds === 1 ? "" : "s"} currently held while someone pays.</p></div>`);
    return (stuck.length || failedEmails.length || due.length || owed.length || left.length ? "" : '<p class="empty">All clear. Nothing needs you right now.</p>') + parts.join("");
  }

  /* Settings (editable) */
  let settings = null;
  const DAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
  const hourLabel = (h) => (h === 0 || h === 24 ? "12 AM" : h === 12 ? "12 PM" : h < 12 ? `${h} AM` : `${h - 12} PM`);

  async function loadSettings() {
    try {
      settings = await api("GET", "/api/admin/settings");
      if (tab === "settings") render();
    } catch (err) { $main.innerHTML = `<p class="empty">${esc(err.message)}</p>`; }
  }

  function renderSettings() {
    if (!settings) { loadSettings(); return '<p class="empty">Loading settings...</p>'; }
    const v = settings.values;
    const opt = (list, cur, label = (x) => x) => list.map((x) => `<option value="${esc(x)}"${x === cur ? " selected" : ""}>${esc(label(x))}</option>`).join("");
    const hours = Array.from({ length: 24 }, (_, i) => i);
    const cals = settings.calendars;
    const calPicker = cals
      ? cals.map((n) => `<label class="check"><input type="checkbox" name="busy" value="${esc(n)}"${v.busyCalendars.includes(n) ? " checked" : ""}> ${esc(n)}</label>`).join("")
      : `<p class="hint">Couldn't reach iCloud to list your calendars right now, so calendar choices can't be changed. Currently: ${esc(v.busyCalendars.join(", "))}.</p>`;
    return `<form class="panel settings" id="settings-form">
      <fieldset><legend>Working hours</legend>
        <div class="row2">
          <label>First start <select name="dayStartHour">${opt(hours, v.dayStartHour, hourLabel)}</select></label>
          <label>Sessions end by <select name="dayEndHour">${opt(hours.map((h) => h + 1), v.dayEndHour, hourLabel)}</select></label>
        </div>
        <p class="hint">In ${settings.timeZone === "America/New_York" ? "Eastern" : esc(settings.timeZone.replace("_", " "))} time. A session can start at any time as long as it ends by the closing hour.</p>
      </fieldset>
      <fieldset><legend>Working days</legend>
        <div class="days">${DAYS.map((d, i) => `<label class="day-chip"><input type="checkbox" name="workDays" value="${i}"${v.workDays.includes(i) ? " checked" : ""}><span>${d}</span></label>`).join("")}</div>
      </fieldset>
      <fieldset><legend>Spacing and notice</legend>
        <div class="row2">
          <label>Buffer around anything on your calendar <select name="bufferMinutes">${opt([0, 5, 10, 15, 20, 30, 45, 60, 90, 120], v.bufferMinutes, (m) => `${m} min`)}</select></label>
          <label>Minimum notice <select name="minNoticeHours">${opt([0, 2, 4, 8, 12, 24, 36, 48, 72, 96, 168], v.minNoticeHours, (h) => (h >= 24 && h % 24 === 0 ? `${h / 24} day${h === 24 ? "" : "s"}` : `${h} hours`))}</select></label>
          <label>Offer start times every <select name="slotStepMinutes">${opt([15, 30, 60], v.slotStepMinutes, (m) => `${m} min`)}</select></label>
        </div>
      </fieldset>
      <fieldset><legend>Calendars</legend>
        <p class="hint">Anything on these calendars blocks booking (events marked "Free" don't).</p>
        <div class="cals">${calPicker}</div>
        ${cals ? `<label>New bookings go into <select name="bookingCalendar">${opt(cals, v.bookingCalendar)}</select></label>
        <p class="hint">The bookings calendar always blocks time too.</p>` : ""}
      </fieldset>
      <fieldset><legend>Bundles and emails</legend>
        <div class="row2">
          <label>Bundles must be used within <select name="packageValidDays">${opt([30, 45, 60, 90, 120, 180, 365], v.packageValidDays, (d) => `${d} days`)}</select></label>
        </div>
        <label class="check"><input type="checkbox" name="remindersEnabled"${v.remindersEnabled ? " checked" : ""}> Send a "finish your booking" email when someone doesn't complete checkout</label>
        <p class="hint">Changing bundle validity applies to bundles bought from now on.</p>
      </fieldset>
      <div class="save-row">
        <p class="hint">${settings.updatedAt ? `Last changed ${esc(fmt(settings.updatedAt, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" }))} by ${esc(settings.updatedBy || "you")}.` : "Using the defaults."} Clients see changes within a minute.</p>
        <div class="row">
          ${settings.updatedAt ? '<button type="button" class="btn" id="settings-reset">Reset to defaults</button>' : ""}
          <button type="submit" class="btn primary">Save settings</button>
        </div>
      </div>
    </form>`;
  }

  async function saveSettings(form) {
    const fd = new FormData(form);
    const body = {
      dayStartHour: Number(fd.get("dayStartHour")),
      dayEndHour: Number(fd.get("dayEndHour")),
      workDays: fd.getAll("workDays").map(Number),
      bufferMinutes: Number(fd.get("bufferMinutes")),
      minNoticeHours: Number(fd.get("minNoticeHours")),
      slotStepMinutes: Number(fd.get("slotStepMinutes")),
      packageValidDays: Number(fd.get("packageValidDays")),
      busyCalendars: settings.calendars ? fd.getAll("busy") : settings.values.busyCalendars,
      bookingCalendar: settings.calendars ? fd.get("bookingCalendar") : settings.values.bookingCalendar,
      remindersEnabled: fd.get("remindersEnabled") === "on",
    };
    const btn = form.querySelector("[type=submit]");
    btn.disabled = true;
    try {
      settings = await api("POST", "/api/admin/settings", body);
      toast("Saved.");
      render();
    } catch (err) { btn.disabled = false; toast(err.message, true); }
  }

  function render() {
    const mode = document.getElementById("mode");
    mode.textContent = data.mode === "test" ? "Test mode" : "Live";
    mode.className = `mode ${data.mode}`;
    renderTabs();
    $main.innerHTML = { upcoming: renderUpcoming, recent: renderRecent, bundles: renderBundles, attention: renderAttention, settings: renderSettings }[tab]?.() ?? renderUpcoming();
  }

  /* Dialogs */
  function openCancel(id) {
    const b = data.bookings.find((x) => x.id === id);
    $dlg.className = "";
    const bundle = b.paid === "bundle";
    $dlg.innerHTML = `<form method="dialog" id="cancel-form">
      <h2 id="dlg-title">Cancel this session?</h2>
      <p>${esc(b.name)} &middot; ${esc(dayLabel(b.start))} at ${esc(clock(b.start))}. ${b.groupId
        ? "They'll be removed from the group session; it goes ahead for everyone else (if they're the last one, the session is cancelled)."
        : "The calendar event and Zoom meeting will be removed and the time opens up again."}</p>
      <label><input type="checkbox" name="notify" checked> Email ${esc(b.name.split(" ")[0])} a cancellation (with a calendar update)</label>
      ${bundle ? '<label><input type="checkbox" name="credit" checked> Return the session to their bundle</label>' : ""}
      ${!bundle && b.paid ? `<label><input type="checkbox" name="refund" checked> Refund ${money(b.paid)} in full now</label>
      <p>Leave this unticked if they'd rather have a new time. They'll be offered either, and you can refund later from here.</p>` : ""}
      ${bundle ? '<label style="display:block">Note for the record (optional)<textarea name="note" maxlength="300"></textarea></label>' : ""}
      <div class="row"><button class="btn" value="keep">Keep it</button><button class="btn danger" value="go">Cancel session</button></div>
    </form>`;
    $dlg.showModal();
    $dlg.querySelector("form").addEventListener("submit", async (e) => {
      if (e.submitter?.value !== "go") return;
      e.preventDefault();
      const f = e.target;
      e.submitter.disabled = true;
      try {
        await api("POST", `/api/admin/bookings/${encodeURIComponent(id)}/cancel`, {
          notifyClient: f.notify.checked, returnCredit: !!f.credit?.checked, refund: !!f.refund?.checked, note: f.note?.value || "",
        });
        $dlg.close();
        toast("Cancelled.");
        load();
      } catch (err) { e.submitter.disabled = false; toast(err.message, true); }
    });
  }

  /* Booking students */

  const HOURS = Array.from({ length: 96 }, (_, i) => `${String(Math.floor(i / 4)).padStart(2, "0")}:${String((i % 4) * 15).padStart(2, "0")}`);
  const hhmmLabel = (hm) => { const [h, m] = hm.split(":").map(Number); return `${h % 12 || 12}:${String(m).padStart(2, "0")} ${h < 12 ? "AM" : "PM"}`; };
  const nyParts = (iso) => {
    const p = Object.fromEntries(new Intl.DateTimeFormat("en-CA", { timeZone: TZ, year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hourCycle: "h23" })
      .formatToParts(Date.parse(iso)).map((x) => [x.type, x.value]));
    return { date: `${p.year}-${p.month}-${p.day}`, time: `${p.hour}:${p.minute}` };
  };
  const whenFields = (date, time) => `
      <label>Date<input type="date" name="date" value="${esc(date)}" required></label>
      <label>Start (Eastern)<select name="time">${HOURS.map((h) => `<option value="${h}"${h === time ? " selected" : ""}>${hhmmLabel(h)}</option>`).join("")}</select></label>`;

  // Warnings for a time: calendar clashes, outside hours, short notice; overlaps block.
  async function checkTime(form, extra) {
    const out = form.querySelector(".warnings");
    try {
      const r = await api("POST", "/api/admin/check-time", { serviceId: form.serviceId?.value || extra.serviceId, date: form.date.value, time: form.time.value, ...extra });
      const w = [];
      if (r.overlapsBooking) w.push('<p class="bad">This overlaps another booking, so it can\'t be booked.</p>');
      if (r.calendarClash && r.calendarClash.length) w.push(`<p>Something on your calendar is at ${r.calendarClash.map((c) => `${esc(clock(c.start))} to ${esc(clock(c.end))}`).join(", ")} (within your buffer). You can still book it.</p>`);
      if (r.outsideHours) w.push("<p>This is outside your usual working hours.</p>");
      if (r.withinNotice) w.push("<p>This is sooner than your usual notice period.</p>");
      out.innerHTML = w.join("");
      return !r.overlapsBooking;
    } catch (err) { out.innerHTML = `<p class="bad">${esc(err.message)}</p>`; return false; }
  }

  let services = null;
  let students = null;
  async function openNewSession() {
    try {
      [services, students] = await Promise.all([
        services ?? fetch("/api/services", { cache: "no-store" }).then((r) => r.json()),
        api("GET", "/api/admin/students"),
      ]);
    } catch (err) { toast(err.message, true); return; }
    const lengths = services.filter((x) => x.kind !== "bundle");
    const priceOf = (id) => lengths.find((x) => x.id === id)?.priceCents ?? 0;
    const tomorrow = nyParts(new Date(Date.now() + 86400000).toISOString()).date;
    $dlg.className = "wide";
    $dlg.innerHTML = `<form method="dialog" id="new-form">
      <h2 id="dlg-title">Book a student</h2>
      <p>One student is a private session; two or more is a group session. Each student gets their own invite, with a link to pay if they owe anything.</p>
      <div class="fgrid">
        <label>Length<select name="serviceId">${lengths.map((x) => `<option value="${esc(x.id)}"${x.id === "coaching-60" ? " selected" : ""}>${x.durationMinutes} min${x.kind === "intro" ? " (intro)" : ""}</option>`).join("")}</select></label>
        ${whenFields(tomorrow, "10:00")}
      </div>
      <div class="warnings"></div>
      <div id="students"></div>
      <p><button type="button" class="btn" id="add-student">+ Add another student</button></p>
      <div class="fgrid">
        <label>Payment deadline<select name="payBy">
          <option value="none">No deadline (reminder only)</option>
          <option value="after24">24 hours after booking</option>
          <option value="after48">48 hours after booking</option>
          <option value="before24">24 hours before the session</option>
        </select></label>
      </div>
      <p class="hint" style="font-size:.75rem;margin:-4px 0 10px">With a deadline, anyone who hasn't paid by then is released automatically and emailed.</p>
      <label class="field">Note in the invite (optional)<textarea name="message" maxlength="1000" placeholder="What to prepare, sides to read..."></textarea></label>
      <div class="row"><button class="btn" value="keep">Never mind</button><button class="btn primary" value="go">Book and send invites</button></div>
    </form>
    <datalist id="known-emails">${students.map((x) => `<option value="${esc(x.email)}">${esc(x.name)}</option>`).join("")}</datalist>`;
    const form = $dlg.querySelector("form");
    const list = form.querySelector("#students");
    const addRow = () => {
      const div = document.createElement("div");
      div.className = "student";
      div.innerHTML = `<button type="button" class="remove" aria-label="Remove this student">&times;</button>
        <div class="fgrid">
          <label>Email<input type="email" name="email" list="known-emails" required autocomplete="off"></label>
          <label>Name<input type="text" name="name" required maxlength="120"></label>
          <label>Pronouns<input type="text" name="pronouns" maxlength="40" placeholder="optional"></label>
          <label>Payment<select name="pay"><option value="price">Pays a price</option></select></label>
          <label class="price">Price ($)<input type="number" name="price" min="0" step="0.01" placeholder="${(priceOf(form.serviceId.value) / 100).toFixed(2)}"></label>
        </div>`;
      list.appendChild(div);
      const email = div.querySelector("[name=email]");
      email.addEventListener("change", () => fillStudent(div));
      div.querySelector("[name=pay]").addEventListener("change", () => { div.querySelector(".price").hidden = div.querySelector("[name=pay]").value !== "price"; });
      div.querySelector(".remove").addEventListener("click", () => { if (list.children.length > 1) div.remove(); });
      email.focus();
    };
    // A known student: fill in their details and offer their bundle.
    const fillStudent = (div) => {
      const known = students.find((x) => x.email.toLowerCase() === div.querySelector("[name=email]").value.trim().toLowerCase());
      const pay = div.querySelector("[name=pay]");
      if (known) {
        if (!div.querySelector("[name=name]").value) div.querySelector("[name=name]").value = known.name;
        if (!div.querySelector("[name=pronouns]").value && known.pronouns) div.querySelector("[name=pronouns]").value = known.pronouns;
      }
      const bundles = known && form.serviceId.value === "coaching-60" ? known.bundles : [];
      pay.innerHTML = '<option value="price">Pays a price</option>'
        + bundles.map((x) => `<option value="${esc(x.id)}">Use bundle (${x.remaining} left)</option>`).join("");
      div.querySelector(".price").hidden = false;
    };
    form.querySelector("#add-student").addEventListener("click", addRow);
    addRow();
    const recheck = () => checkTime(form, {});
    form.serviceId.addEventListener("change", () => {
      for (const d of list.children) { d.querySelector("[name=price]").placeholder = (priceOf(form.serviceId.value) / 100).toFixed(2); fillStudent(d); }
      recheck();
    });
    form.date.addEventListener("change", recheck);
    form.time.addEventListener("change", recheck);
    recheck();
    $dlg.showModal();
    form.addEventListener("submit", async (e) => {
      if (e.submitter?.value !== "go") return;
      e.preventDefault();
      if (!form.reportValidity()) return;
      const body = {
        serviceId: form.serviceId.value, date: form.date.value, time: form.time.value,
        payBy: form.payBy.value, message: form.message.value,
        students: [...list.children].map((d) => {
          const pay = d.querySelector("[name=pay]").value;
          const price = d.querySelector("[name=price]").value;
          return {
            name: d.querySelector("[name=name]").value, email: d.querySelector("[name=email]").value.trim(),
            pronouns: d.querySelector("[name=pronouns]").value,
            ...(pay === "price" ? (price === "" ? {} : { priceCents: Math.round(Number(price) * 100) }) : { packageId: pay }),
          };
        }),
      };
      e.submitter.disabled = true;
      try {
        await api("POST", "/api/admin/sessions", body);
        $dlg.close();
        toast(body.students.length > 1 ? "Group session booked. Invites sent." : "Booked. Invite sent.");
        load();
      } catch (err) { e.submitter.disabled = false; toast(err.message, true); }
    });
  }

  function openMove(target, startIso, mins) {
    const now = nyParts(startIso);
    $dlg.className = "";
    $dlg.innerHTML = `<form method="dialog">
      <h2 id="dlg-title">Move this session</h2>
      <p>Currently ${esc(dayLabel(startIso))} at ${esc(clock(startIso))} (${mins} min). ${target.groupId ? "Every student" : "The student"} gets an updated invite.</p>
      <div class="fgrid">${whenFields(now.date, now.time)}</div>
      <div class="warnings"></div>
      <div class="row"><button class="btn" value="keep">Never mind</button><button class="btn primary" value="go">Move it</button></div>
    </form>`;
    const form = $dlg.querySelector("form");
    const b = data.bookings.find((x) => (target.groupId ? x.groupId === target.groupId : x.id === target.bookingId));
    const extra = { serviceId: b.serviceId, ...(target.groupId ? { movingGroupId: target.groupId } : { movingBookingId: target.bookingId }) };
    const recheck = () => checkTime(form, extra);
    form.date.addEventListener("change", recheck);
    form.time.addEventListener("change", recheck);
    $dlg.showModal();
    form.addEventListener("submit", async (e) => {
      if (e.submitter?.value !== "go") return;
      e.preventDefault();
      e.submitter.disabled = true;
      try {
        const path = target.groupId ? `/api/admin/groups/${encodeURIComponent(target.groupId)}/move` : `/api/admin/bookings/${encodeURIComponent(target.bookingId)}/move`;
        await api("POST", path, { date: form.date.value, time: form.time.value });
        $dlg.close();
        toast("Moved. Updated invites sent.");
        load();
      } catch (err) { e.submitter.disabled = false; toast(err.message, true); }
    });
  }

  function openCancelGroup(groupId) {
    const seats = data.bookings.filter((x) => x.groupId === groupId && x.status === "confirmed");
    const paidCount = seats.filter((x) => x.paid && x.paid !== "bundle").length;
    $dlg.className = "";
    $dlg.innerHTML = `<form method="dialog">
      <h2 id="dlg-title">Cancel this group session?</h2>
      <p>${esc(dayLabel(seats[0].start))} at ${esc(clock(seats[0].start))} &middot; ${seats.length} student${seats.length === 1 ? "" : "s"}. The calendar event and Zoom meeting will be removed and the time opens up again. Bundle credits go back.</p>
      <label><input type="checkbox" name="notify" checked> Email everyone a cancellation</label>
      ${paidCount ? `<label><input type="checkbox" name="refund" checked> Refund the ${paidCount} student${paidCount === 1 ? "" : "s"} who paid, in full</label>` : ""}
      <div class="row"><button class="btn" value="keep">Keep it</button><button class="btn danger" value="go">Cancel session</button></div>
    </form>`;
    $dlg.showModal();
    $dlg.querySelector("form").addEventListener("submit", async (e) => {
      if (e.submitter?.value !== "go") return;
      e.preventDefault();
      e.submitter.disabled = true;
      try {
        await api("POST", `/api/admin/groups/${encodeURIComponent(groupId)}/cancel`, { notifyClient: e.target.notify.checked, refund: !!e.target.refund?.checked });
        $dlg.close();
        toast("Group session cancelled.");
        load();
      } catch (err) { e.submitter.disabled = false; toast(err.message, true); }
    });
  }

  async function resend(id, btn) {
    btn.disabled = true;
    try { await api("POST", `/api/admin/bookings/${encodeURIComponent(id)}/resend`); toast("Invite sent again."); }
    catch (err) { toast(err.message, true); }
    btn.disabled = false;
  }

  document.getElementById("new-session").addEventListener("click", openNewSession);

  async function refundNow(id, btn) {
    const b = data.bookings.find((x) => x.id === id);
    if (!confirm(`Refund ${money(b.paid)} to ${b.name} in full?`)) return;
    btn.disabled = true;
    try {
      const r = await api("POST", `/api/admin/bookings/${encodeURIComponent(id)}/refund`);
      toast(r.message || "Refunded.", !r.ok);
      load();
    } catch (err) { btn.disabled = false; toast(err.message, true); }
  }

  function openCredit(id, delta) {
    const p = data.packages.find((x) => x.id === id);
    $dlg.className = "";
    $dlg.innerHTML = `<form method="dialog">
      <h2 id="dlg-title">${delta > 0 ? "Add a session" : "Remove a session"}</h2>
      <p>${esc(p.name)} &middot; ${esc(p.bundle)} &middot; ${p.remaining} left now.</p>
      <label style="display:block">Reason (for the record)<textarea name="note" maxlength="300" placeholder="${delta > 0 ? "Makeup for a session I had to move" : "Late cancellation"}"></textarea></label>
      <div class="row"><button class="btn" value="keep">Never mind</button><button class="btn primary" value="go">${delta > 0 ? "Add 1" : "Remove 1"}</button></div>
    </form>`;
    $dlg.showModal();
    $dlg.querySelector("form").addEventListener("submit", async (e) => {
      if (e.submitter?.value !== "go") return;
      e.preventDefault();
      e.submitter.disabled = true;
      try {
        await api("POST", `/api/admin/packages/${encodeURIComponent(id)}/credits`, { delta, note: e.target.note.value });
        $dlg.close();
        toast(delta > 0 ? "Session added." : "Session removed.");
        load();
      } catch (err) { e.submitter.disabled = false; toast(err.message, true); }
    });
  }

  function openExtend(id) {
    const p = data.packages.find((x) => x.id === id);
    $dlg.className = "";
    $dlg.innerHTML = `<form method="dialog">
      <h2 id="dlg-title">Extend use-by date</h2>
      <p>${esc(p.name)} &middot; currently ${esc(p.expiresAt ? fmt(p.expiresAt, { month: "long", day: "numeric" }) : "")}.</p>
      <div class="row" style="justify-content:flex-start">
        <button class="btn" value="7">+7 days</button><button class="btn" value="14">+14 days</button><button class="btn" value="30">+30 days</button>
      </div>
      <div class="row"><button class="btn" value="keep">Never mind</button></div>
    </form>`;
    $dlg.showModal();
    $dlg.querySelector("form").addEventListener("submit", async (e) => {
      const days = Number(e.submitter?.value);
      if (!days) return;
      e.preventDefault();
      try {
        await api("POST", `/api/admin/packages/${encodeURIComponent(id)}/extend`, { days });
        $dlg.close();
        toast(`Extended by ${days} days.`);
        load();
      } catch (err) { toast(err.message, true); }
    });
  }

  document.addEventListener("click", (e) => {
    const t = e.target.closest("[data-tab],[data-cancel],[data-credit],[data-extend],[data-refund],[data-move],[data-move-group],[data-cancel-group],[data-resend]");
    if (!t) return;
    if (t.dataset.tab) { tab = t.dataset.tab; history.replaceState(null, "", `#${tab}`); render(); }
    else if (t.dataset.cancel) openCancel(t.dataset.cancel);
    else if (t.dataset.credit) openCredit(t.dataset.credit, Number(t.dataset.delta));
    else if (t.dataset.extend) openExtend(t.dataset.extend);
    else if (t.dataset.refund) refundNow(t.dataset.refund, t);
    else if (t.dataset.move) { const b = data.bookings.find((x) => x.id === t.dataset.move); openMove({ bookingId: b.id }, b.start, minutes(b)); }
    else if (t.dataset.moveGroup) { const b = data.bookings.find((x) => x.groupId === t.dataset.moveGroup); openMove({ groupId: b.groupId }, b.start, minutes(b)); }
    else if (t.dataset.cancelGroup) openCancelGroup(t.dataset.cancelGroup);
    else if (t.dataset.resend) resend(t.dataset.resend, t);
  });

  document.addEventListener("submit", (e) => {
    if (e.target.id === "settings-form") { e.preventDefault(); saveSettings(e.target); }
  });
  document.addEventListener("click", async (e) => {
    if (e.target.id !== "settings-reset") return;
    if (!confirm("Go back to the default settings?")) return;
    try { settings = await api("POST", "/api/admin/settings/reset"); toast("Back to defaults."); render(); } catch (err) { toast(err.message, true); }
  });

  load();
  setInterval(() => { if (!$dlg.open && tab !== "settings" && document.visibilityState === "visible") load(); }, 60000);
})();
