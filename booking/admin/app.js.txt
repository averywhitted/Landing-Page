// Served as JavaScript at /admin/app.js (stored as .txt so it bundles into the service as text).
// Booking admin (served at book.averywhitted.com/admin behind Cloudflare Access).
(() => {
  "use strict";
  const TZ = "America/New_York";
  const $main = document.getElementById("main");
  const $tabs = document.getElementById("tabs");
  const $dlg = document.getElementById("dlg");
  const store = {
    get(k) { try { return localStorage.getItem(k); } catch { return null; } },
    set(k, v) { try { localStorage.setItem(k, v); } catch { /* private mode */ } },
  };
  let data = null;
  let tab = (location.hash || "#sessions").slice(1);
  if (tab === "upcoming" || tab === "recent") tab = "sessions";
  let view = store.get("admin-view") === "calendar" ? "calendar" : "list";
  let scale = ["day", "week", "month"].includes(store.get("admin-scale")) ? store.get("admin-scale") : "week";
  let calDate = null;           // "YYYY-MM-DD" the calendar is showing (Eastern)
  const calData = new Map();    // range -> { bookings, busy }
  let filters = { past: false, cancelled: false };
  try { filters = { ...filters, ...JSON.parse(store.get("admin-filters") || "{}") }; } catch { /* default */ }
  let studentOpen = null;       // the student whose details are open (dialogs return there)
  let studentFilter = "";
  const known = new Map();      // booking id -> booking, from the overview and student details

  const esc = (s) => String(s ?? "").replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
  const money = (c) => (c % 100 ? `$${(c / 100).toFixed(2)}` : `$${c / 100}`);
  const fmt = (iso, o) => new Intl.DateTimeFormat("en-US", { timeZone: TZ, ...o }).format(Date.parse(iso));
  const dayLabel = (iso) => fmt(iso, { weekday: "long", month: "long", day: "numeric" });
  const shortDate = (iso) => fmt(iso, { month: "short", day: "numeric", year: "numeric" });
  const clock = (iso) => fmt(iso, { hour: "numeric", minute: "2-digit" });
  const minutes = (b) => Math.round((Date.parse(b.end) - Date.parse(b.start)) / 60000);
  const nyParts = (iso) => {
    const p = Object.fromEntries(new Intl.DateTimeFormat("en-CA", { timeZone: TZ, year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", hourCycle: "h23" })
      .formatToParts(Date.parse(iso)).map((x) => [x.type, x.value]));
    return { date: `${p.year}-${p.month}-${p.day}`, time: `${p.hour}:${p.minute}` };
  };

  // A student's name, always with their pronouns right after it. With an id it
  // opens their profile.
  const nameTag = (name, pronouns, customerId) => {
    const inner = `${esc(name)}${pronouns ? ` <span class="pn">(${esc(pronouns)})</span>` : ""}`;
    return customerId
      ? `<button type="button" class="name-link" data-student="${esc(customerId)}" title="Open ${esc(name)}'s profile">${inner}</button>`
      : `<span class="name">${inner}</span>`;
  };
  const firstName = (name) => String(name || "").split(/\s+/)[0];

  // A button with a hover explanation.
  const btn = (label, attrs, tip, kind = "") =>
    `<button type="button" class="btn${kind ? ` ${kind}` : ""}" ${attrs} title="${esc(tip)}" aria-label="${esc(`${label}: ${tip}`)}">${label}</button>`;
  const link = (label, href, tip, kind = "") =>
    `<a class="btn${kind ? ` ${kind}` : ""}" href="${esc(href)}" target="_blank" rel="noopener" title="${esc(tip)}">${label}</a>`;

  async function api(method, path, body) {
    const r = await fetch(path, {
      method,
      headers: { "X-Admin": "1", ...(body ? { "Content-Type": "application/json" } : {}) },
      body: body ? JSON.stringify(body) : undefined,
      cache: "no-store",
      credentials: "same-origin",
    });
    const out = await r.json().catch(() => ({}));
    if (!r.ok) throw new Error(out.error || `Request failed (${r.status})`);
    return out;
  }

  function toast(text, isErr = false) {
    const t = document.createElement("div");
    t.className = `toast${isErr ? " err" : ""}`;
    t.setAttribute("role", isErr ? "alert" : "status");
    t.textContent = text;
    document.body.appendChild(t);
    setTimeout(() => t.remove(), 4500);
  }

  async function load() {
    try {
      data = await api("GET", "/api/admin/overview");
      for (const b of data.bookings) known.set(b.id, b);
      render();
    } catch (err) {
      $main.innerHTML = `<p class="empty">${esc(err.message)}</p>`;
    }
  }

  // After an action: refresh now, and again shortly (refunds and emails finish in the background).
  async function afterAction(message) {
    if (message) toast(message);
    calData.clear();
    await load();
    if (studentOpen) openStudent(studentOpen); else if ($dlg.open) $dlg.close();
    setTimeout(load, 4000);
  }

  // Dialogs share one <dialog>. Cancelling a dialog opened from a student's
  // details goes back to those details.
  function showDialog(html, cls = "") {
    $dlg.className = cls;
    $dlg.innerHTML = html;
    document.documentElement.classList.add("dlg-open");
    if (!$dlg.open) $dlg.showModal();
    $dlg.scrollTop = 0;
    for (const b of $dlg.querySelectorAll("[data-back]")) b.addEventListener("click", back);
  }
  function back() { if (studentOpen) openStudent(studentOpen); else $dlg.close(); }
  $dlg.addEventListener("close", () => { studentOpen = null; document.documentElement.classList.remove("dlg-open"); });
  const cancelButton = (label = "Cancel") => `<button type="button" class="btn" data-back>${label}</button>`;

  /* ── Tabs ── */

  const refundsDue = () => data.packages.filter((p) => p.cancelled && p.refundDueCents && !p.refunded);
  const stuckRefunds = () => data.bookings.filter((b) => b.refundStuck || b.refundOwed === "choice");
  function problemsCount() {
    const p = data.problems;
    const dueNow = data.bookings.filter((b) => b.dueCents && b.status === "confirmed" && !b.past).length;
    return p.failedEmails.length + p.stuck.length + refundsDue().length + stuckRefunds().length + p.leftOnCalendar.length + p.zoomLeft.length + dueNow;
  }

  function renderTabs() {
    // A group session counts once, however many students it has.
    const upcoming = new Set(data.bookings.filter((b) => b.status === "confirmed" && !b.past).map((b) => b.groupId || b.id)).size;
    const tabs = [
      ["sessions", "Sessions", upcoming],
      ["students", "Students", null],
      ["bundles", "Bundles", data.packages.filter((p) => !p.cancelled).length],
      ["attention", "Needs attention", problemsCount()],
      ["settings", "Settings", null],
    ];
    $tabs.innerHTML = tabs.map(([id, label, n]) => `<button class="tab" role="tab" aria-selected="${id === tab}" data-tab="${id}">${label}${n !== null ? `<span class="count${id === "attention" && n ? " alert" : ""}">${n}</span>` : ""}</button>`).join("");
  }

  /* ── Session cards ── */

  // How a session is paid for: "$75 paid", "$75 due by Oct 1", "Bundle credit", "Free".
  const payLabel = (b) => b.paid === "bundle" ? "Bundle credit"
    : b.paid ? `${money(b.paid)} paid${b.promoCode ? ` (code ${b.promoCode})` : ""}`
    : b.dueCents ? `${money(b.dueCents)} due${b.payBy ? ` by ${fmt(b.payBy, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" })}` : ""}`
    : "Free";
  const payChip = (b) => `<span class="chip${b.dueCents ? " bad" : b.paid && b.paid !== "bundle" ? " ok" : ""}">${esc(payLabel(b))}</span>`;
  const statusChip = (b) => b.status !== "cancelled" ? ""
    : `<span class="chip bad">${b.unpaidReleased ? "Released (unpaid)" : b.cancelReason === "bundle_cancelled" ? "Cancelled with bundle"
      : `Cancelled${b.cancelReason === "avery_cancelled" ? " by you" : b.cancelReason === "client_cancelled" ? " by student" : ""}`}</span>`;
  const refundChip = (b) => b.refunded ? '<span class="chip ok">Refunded</span>'
    : b.refundOwed === "auto" ? `<span class="chip${b.refundStuck ? " bad" : ""}">${b.refundStuck ? "Refund stuck" : "Refund in progress"}</span>`
    : b.refundOwed === "choice" ? '<span class="chip">Not refunded</span>' : "";

  const remindBtn = (b, small) => b.dueCents && b.status === "confirmed" && !b.past
    ? btn(small ? "Remind" : "Send payment reminder", `data-remind="${esc(b.id)}"`, `Email ${firstName(b.name)} a reminder to pay ${money(b.dueCents)}, with the payment link`, small ? "sm" : "")
    : "";

  function bookingCard(b) {
    const live = b.status === "confirmed" && !b.past;
    const chips = [
      payChip(b), statusChip(b), refundChip(b),
      b.byAvery ? '<span class="chip">Booked by you</span>' : "",
      live ? `<span class="chip ${b.inCalendar ? "ok" : "bad"}">${b.inCalendar ? "In calendar" : "Not in calendar"}</span>` : "",
      live && !b.emailed ? '<span class="chip bad">Not emailed</span>' : "",
    ].join("");
    const intake = b.intake ? Object.entries({ Goal: b.intake.goal, Material: b.intake.material, Link: b.intake.link, Notes: b.intake.notes })
      .filter(([, v]) => v).map(([k, v]) => `<dt>${k}</dt><dd>${k === "Link" ? `<a href="${esc(v)}" target="_blank" rel="noopener">${esc(v)}</a>` : esc(v)}</dd>`).join("") : "";
    const actions = [
      b.zoomUrl && live ? link("Join Zoom", b.zoomUrl, "Open this session's Zoom meeting", "primary") : "",
      remindBtn(b),
      b.stripeUrl ? link("Payment", b.stripeUrl, "Open this payment in your Stripe dashboard") : "",
      live ? btn("Change time", `data-move="${esc(b.id)}"`, "Move this session to another date or time. The student gets an updated invite.") : "",
      live ? btn("Resend invite", `data-resend="${esc(b.id)}"`, "Email the student their invite again (with the payment link if they still owe)") : "",
      b.refundOwed === "choice" ? btn(`Refund ${money(b.paid)}`, `data-refund="${esc(b.id)}"`, "Refund this cancelled session in full") : "",
      live ? btn("Cancel session", `data-cancel="${esc(b.id)}"`, "Cancel this session. You'll choose whether to refund and whether to email the student.", "danger") : "",
    ].filter(Boolean).join("");
    return `<div class="card${live ? "" : " muted"}">
      <div class="time">${esc(clock(b.start))}<small>${minutes(b)} min</small></div>
      <div class="who">
        <strong>${nameTag(b.name, b.pronouns, b.customerId)}</strong>
        <div class="sub"><a href="mailto:${esc(b.email)}">${esc(b.email)}</a> &middot; ${esc(b.service)}</div>
        <div class="chips">${chips}</div>
        ${intake ? `<details class="intake"><summary>Intake answers</summary><dl>${intake}</dl></details>` : ""}
      </div>
      ${actions ? `<div class="actions">${actions}</div>` : ""}
    </div>`;
  }

  function groupCard(seats) {
    const first = seats[0];
    const active = seats.filter((s) => s.status === "confirmed");
    const live = active.length && !first.past;
    const rows = seats.map((s) => `<div class="seat">
        <div class="who-line"><strong>${nameTag(s.name, s.pronouns, s.customerId)}</strong><small><a href="mailto:${esc(s.email)}">${esc(s.email)}</a></small></div>
        <span class="seat-chips">${payChip(s)}${statusChip(s)}${refundChip(s)}</span>
        <span class="seat-actions">
          ${remindBtn(s, true)}
          ${s.refundOwed === "choice" ? btn(`Refund ${money(s.paid)}`, `data-refund="${esc(s.id)}"`, "Refund this student in full", "sm") : ""}
          ${live && s.status === "confirmed" ? btn("Resend", `data-resend="${esc(s.id)}"`, "Email this student their invite again", "sm")
            + btn("Remove", `data-cancel="${esc(s.id)}"`, "Take this student out of the group. You'll choose whether to refund them.", "sm danger") : ""}
        </span>
      </div>`).join("");
    return `<div class="card${live ? "" : " muted"}">
      <div class="time">${esc(clock(first.start))}<small>${minutes(first)} min</small></div>
      <div class="who">
        <strong>Group session</strong> <span class="sub" style="display:inline">&middot; ${active.length} of ${seats.length} student${seats.length === 1 ? "" : "s"}</span>
        ${live ? `<div class="chips"><span class="chip ${first.inCalendar ? "ok" : "bad"}">${first.inCalendar ? "In calendar" : "Not in calendar"}</span></div>` : ""}
        <div class="seats">${rows}</div>
      </div>
      ${live ? `<div class="actions">
        ${first.zoomUrl ? link("Join Zoom", first.zoomUrl, "Open this session's Zoom meeting", "primary") : ""}
        ${btn("Change time", `data-move="${esc(first.id)}"`, "Move the whole session. Every student gets an updated invite.")}
        ${btn("Cancel session", `data-cancel-group="${esc(first.groupId)}"`, "Cancel the session for everyone. You'll choose whether to refund those who paid.", "danger")}
      </div>` : ""}
    </div>`;
  }

  // Sessions in order, with students in the same group session as one entry.
  function sessionsOf(list) {
    const items = [];
    const groups = new Map();
    for (const b of list) {
      if (!b.groupId) { items.push({ start: b.start, one: b }); continue; }
      if (!groups.has(b.groupId)) { const g = { start: b.start, seats: [] }; groups.set(b.groupId, g); items.push(g); }
      groups.get(b.groupId).seats.push(b);
    }
    return items;
  }
  const cardFor = (it) => (it.one ? bookingCard(it.one) : groupCard(it.seats));

  function byDay(list) {
    let out = "";
    let last = "";
    for (const it of sessionsOf(list)) {
      const d = dayLabel(it.start);
      if (d !== last) { out += `<p class="day">${esc(d)}</p>`; last = d; }
      out += cardFor(it);
    }
    return out;
  }

  // Filters shared by the list and the calendar.
  const visible = (b) => (filters.cancelled || b.status === "confirmed") && (filters.past || !b.past);

  function toolbar() {
    const seg = (name, items, cur) => `<div class="seg" role="group" aria-label="${name}">${items.map(([v, label, tip]) =>
      `<button type="button" data-${name}="${v}" aria-pressed="${cur === v}" title="${esc(tip)}">${label}</button>`).join("")}</div>`;
    const upcomingOnly = !filters.past && !filters.cancelled;
    const all = filters.past && filters.cancelled;
    return `<div class="toolbar">
      <div class="filters">
        <button type="button" class="fchip" data-filter="upcoming" aria-pressed="${upcomingOnly}" title="Only sessions that haven't happened yet">Upcoming</button>
        <button type="button" class="fchip" data-filter="all" aria-pressed="${all}" title="Everything, including past and cancelled sessions">All</button>
        <span class="fsep"></span>
        <button type="button" class="fchip" data-filter="past" aria-pressed="${filters.past}" title="Show sessions that have already happened">Past</button>
        <button type="button" class="fchip" data-filter="cancelled" aria-pressed="${filters.cancelled}" title="Show cancelled sessions">Cancelled</button>
      </div>
      <div class="views">
        ${view === "calendar" ? seg("scale", [["day", "Day", "One day"], ["week", "Week", "One week"], ["month", "Month", "A month at a time"]], scale) : ""}
        ${seg("view", [["list", "List", "Sessions as a list"], ["calendar", "Calendar", "Sessions on a calendar"]], view)}
      </div>
    </div>`;
  }

  function renderSessions() {
    if (view === "calendar") return toolbar() + renderCalendar();
    const list = data.bookings.filter(visible);
    const note = filters.past ? '<p class="hint">The list covers the last 30 days and the next 4 months. Use the calendar for further back or ahead.</p>' : "";
    return toolbar() + note + (list.length ? byDay(list) : `<p class="empty">No ${filters.past || filters.cancelled ? "" : "upcoming "}sessions to show.</p>`);
  }

  /* ── Calendar (day, week, month), with blocked time from your calendars ── */

  const todayNY = () => nyParts(new Date().toISOString()).date;
  const addDaysTo = (date, n) => { const [y, m, d] = date.split("-").map(Number); return new Date(Date.UTC(y, m - 1, d + n)).toISOString().slice(0, 10); };
  const weekdayOf = (date) => { const [y, m, d] = date.split("-").map(Number); return new Date(Date.UTC(y, m - 1, d)).getUTCDay(); };
  const minsOf = (iso) => { const [h, m] = nyParts(iso).time.split(":").map(Number); return h * 60 + m; };
  const hourText = (mins) => { const h = Math.floor(mins / 60) % 24; return `${h % 12 || 12}${h < 12 ? "am" : "pm"}`; };

  // The dates on screen for the current scale.
  function calRange() {
    if (!calDate) calDate = todayNY();
    if (scale === "day") return { dates: [calDate], first: calDate };
    if (scale === "week") {
      const start = addDaysTo(calDate, -weekdayOf(calDate));
      return { dates: Array.from({ length: 7 }, (_, i) => addDaysTo(start, i)), first: start };
    }
    const monthStart = `${calDate.slice(0, 7)}-01`;
    const gridStart = addDaysTo(monthStart, -weekdayOf(monthStart));
    const [y, m] = calDate.split("-").map(Number);
    const monthEnd = new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10);
    const gridEnd = addDaysTo(monthEnd, 6 - weekdayOf(monthEnd));
    const dates = [];
    for (let d = gridStart; d <= gridEnd; d = addDaysTo(d, 1)) dates.push(d);
    return { dates, first: gridStart, month: calDate.slice(0, 7) };
  }

  function calendarData(dates) {
    const from = `${addDaysTo(dates[0], -1)}T00:00:00Z`;
    const to = `${addDaysTo(dates.at(-1), 2)}T00:00:00Z`;
    const key = `${from}|${to}`;
    if (calData.has(key)) return calData.get(key);
    calData.set(key, null);
    api("GET", `/api/admin/calendar?from=${encodeURIComponent(from)}&to=${encodeURIComponent(to)}`)
      .then((r) => { for (const b of r.bookings) known.set(b.id, b); calData.set(key, r); if (tab === "sessions" && view === "calendar") render(); })
      .catch((err) => { calData.delete(key); toast(err.message, true); });
    return null;
  }

  // Busy time split into per-day pieces: { date, start, end } in minutes.
  function busyPieces(busy) {
    const out = [];
    for (const b of busy || []) {
      const endIso = new Date(Date.parse(b.end) - 60000).toISOString();
      const lastDate = nyParts(endIso).date;
      for (let d = nyParts(b.start).date; d <= lastDate; d = addDaysTo(d, 1)) {
        const start = d === nyParts(b.start).date ? minsOf(b.start) : 0;
        const end = d === lastDate ? minsOf(endIso) + 1 : 1440;
        if (end > start) out.push({ date: d, start, end });
      }
    }
    return out;
  }

  function renderCalendar() {
    const { dates, month } = calRange();
    const loaded = calendarData(dates);
    const items = loaded ? sessionsOf(loaded.bookings.filter(visible)) : [];
    const byDate = new Map();
    for (const it of items) {
      const d = nyParts(it.start).date;
      if (!byDate.has(d)) byDate.set(d, []);
      byDate.get(d).push(it);
    }
    const busy = loaded ? busyPieces(loaded.busy) : [];
    const today = todayNY();
    const label = (it) => {
      const b = it.one || it.seats[0];
      const due = it.one ? b.dueCents : it.seats.some((x) => x.dueCents && x.status === "confirmed");
      const cancelled = it.one ? b.status === "cancelled" : it.seats.every((x) => x.status === "cancelled");
      const who = it.one ? b.name : `Group (${it.seats.filter((x) => x.status === "confirmed").length})`;
      const tip = `${clock(b.start)} to ${clock(b.end)}: ${it.one ? `${b.name}${b.pronouns ? ` (${b.pronouns})` : ""}` : "Group session"}${due ? ", payment due" : ""}${cancelled ? ", cancelled" : ""}`;
      return { b, due, cancelled, who, tip };
    };
    const title = scale === "month"
      ? new Intl.DateTimeFormat("en-US", { month: "long", year: "numeric", timeZone: "UTC" }).format(Date.parse(`${month}-01T12:00:00Z`))
      : scale === "week"
        ? `${fmt(`${dates[0]}T12:00:00Z`, { month: "short", day: "numeric" })} to ${fmt(`${dates[6]}T12:00:00Z`, { month: "short", day: "numeric", year: "numeric" })}`
        : fmt(`${dates[0]}T12:00:00Z`, { weekday: "long", month: "long", day: "numeric", year: "numeric" });
    const head = `<div class="cal-head">
        ${btn("&lsaquo;", 'data-cal="-1"', `Previous ${scale}`)}
        <strong>${esc(title)}</strong>
        ${btn("&rsaquo;", 'data-cal="1"', `Next ${scale}`)}
        ${btn("Today", 'data-cal="0"', "Back to today")}
        ${loaded ? "" : '<span class="hint">Loading...</span>'}
      </div>`;
    const legend = `<p class="cal-legend"><span class="lg session"></span>Session <span class="lg due"></span>Payment due <span class="lg busy"></span>Blocked on your calendars${loaded && loaded.busy === null ? " (couldn't reach iCloud)" : ""}</p>`;

    if (scale === "month") {
      const cells = dates.map((d) => {
        const its = byDate.get(d) || [];
        const bz = busy.filter((x) => x.date === d);
        return `<div class="cal-cell${d === today ? " today" : ""}${d < today ? " past" : ""}${d.slice(0, 7) !== month ? " other" : ""}">
          <button type="button" class="cal-num" data-day="${d}" title="Open this day">${Number(d.slice(8))}</button>
          ${its.map((it) => { const x = label(it); return `<button type="button" class="cal-item${x.due ? " due" : ""}${x.cancelled ? " cancelled" : ""}" data-session="${esc(x.b.id)}" title="${esc(x.tip)}">${esc(hourText(minsOf(x.b.start)))} ${esc(x.who.split(" ")[0])}</button>`; }).join("")}
          ${bz.slice(0, 2).map((x) => `<span class="cal-busy" title="Blocked on your calendars">${x.start === 0 && x.end === 1440 ? "Blocked all day" : `Busy ${hourText(x.start)} to ${hourText(x.end)}`}</span>`).join("")}${bz.length > 2 ? `<span class="cal-busy">+${bz.length - 2} more</span>` : ""}
        </div>`;
      }).join("");
      return `<div class="panel cal">${head}
        <div class="cal-grid">${["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"].map((d) => `<div class="cal-dow">${d}</div>`).join("")}${cells}</div>
        ${legend}</div>`;
    }

    // Day and week: a time grid. It covers 7am to 10pm, stretched to fit any session.
    let lo = 7 * 60, hi = 22 * 60;
    for (const it of items) { lo = Math.min(lo, Math.floor(minsOf(it.start) / 60) * 60); const e = minsOf((it.one || it.seats[0]).end); hi = Math.max(hi, Math.ceil((e || 1440) / 60) * 60); }
    const PX = 0.8; // pixels per minute
    const hours = [];
    for (let m = lo; m < hi; m += 60) hours.push(m);
    const cols = dates.map((d) => {
      const bz = busy.filter((x) => x.date === d && x.end > lo && x.start < hi).map((x) => {
        const top = (Math.max(x.start, lo) - lo) * PX, h = (Math.min(x.end, hi) - Math.max(x.start, lo)) * PX;
        return `<div class="tg-busy" style="top:${top}px;height:${h}px" title="Blocked on your calendars (${hourText(x.start)} to ${x.end === 1440 ? "midnight" : hourText(x.end)})"></div>`;
      }).join("");
      const ss = (byDate.get(d) || []).map((it) => {
        const x = label(it);
        const s0 = minsOf(x.b.start), e0 = minsOf(x.b.end) || 1440;
        return `<button type="button" class="tg-item${x.due ? " due" : ""}${x.cancelled ? " cancelled" : ""}" style="top:${(s0 - lo) * PX}px;height:${Math.max(18, (e0 - s0) * PX - 2)}px" data-session="${esc(x.b.id)}" title="${esc(x.tip)}">
          <strong>${esc(clock(x.b.start))}</strong> ${esc(x.who)}</button>`;
      }).join("");
      return `<div class="tg-col${d === today ? " today" : ""}" style="height:${(hi - lo) * PX}px">${bz}${ss}</div>`;
    }).join("");
    return `<div class="panel cal">${head}
      <div class="tg" style="--cols:${dates.length}">
        <div class="tg-head"><div></div>${dates.map((d) => `<button type="button" class="tg-day${d === today ? " today" : ""}" data-day="${d}" title="Open this day">${esc(fmt(`${d}T12:00:00Z`, { weekday: "short" }))} <b>${Number(d.slice(8))}</b></button>`).join("")}</div>
        <div class="tg-body"><div class="tg-times">${hours.map((m) => `<span style="height:${60 * PX}px">${hourText(m)}</span>`).join("")}</div>${cols}</div>
      </div>
      ${legend}</div>`;
  }

      return '<p class="empty">Loading students...</p>';
    }
    const q = studentFilter.trim().toLowerCase();
    const sixtyDaysAgo = new Date(Date.now() - 60 * 86400000).toISOString();
    const shows = {
      all: ["All", () => true],
      upcoming: ["Has upcoming", (s) => s.upcoming > 0],
      owes: ["Owes money", (s) => s.owedCents > 0],
      bundle: ["Has a bundle", (s) => s.bundles.length > 0],
      quiet: ["Quiet for 60+ days", (s) => !s.upcoming && (!s.lastStart || s.lastStart < sixtyDaysAgo)],
    };
    const keys = {
      name: (s) => s.name.toLowerCase(),
      next: (s) => s.nextStart || "9999",

      owes: (s) => s.owedCents,
      bundle: (s) => s.bundles.reduce((n, x) => n + x.remaining, 0),
    };
    const list = students
      .filter((s) => !q || s.name.toLowerCase().includes(q) || s.email.toLowerCase().includes(q))

  async function openStudent(id) {

    let s;
    try { s = await api("GET", `/api/admin/students/${encodeURIComponent(id)}`); }
    catch (err) { toast(err.message, true); return; }
    if (studentOpen !== id) return;
    for (const x of s.sessions) known.set(x.id, { ...x, name: s.name, email: s.email, pronouns: s.pronouns, customerId: s.id, past: x.status === "past", status: x.status === "cancelled" ? "cancelled" : "confirmed", unpaidReleased: x.cancelReason === "unpaid" });
    const upcoming = s.sessions.filter((x) => x.status === "upcoming").reverse();
    const past = s.sessions.filter((x) => x.status !== "upcoming");
    const owed = upcoming.reduce((n, x) => n + (x.dueCents || 0), 0);
    const sessionRow = (x) => {
      const k = known.get(x.id);
      return `<div class="seat">
        <div class="who-line"><strong>${esc(fmt(x.start, { weekday: "short", month: "short", day: "numeric" }))}, ${esc(clock(x.start))}</strong>

          ${btn("Change time", `data-move="${esc(x.id)}"`, x.group ? "Move the whole group session" : "Move this session. They get an updated invite.", "sm")}
          ${btn("Resend", `data-resend="${esc(x.id)}"`, "Email them their invite again", "sm")}
          ${btn(x.group ? "Remove" : "Cancel", `data-cancel="${esc(x.id)}"`, x.group ? "Take them out of the group session" : "Cancel this session", "sm danger")}
        </span>` : ""}
      </div>`;
    };
    const bundleRow = (p) => {
      const live = p.status === "active" && !p.expired;
      const state = p.status === "cancelled"
        ? `Cancelled${p.cancelReason === "avery_cancelled" ? " by you" : ""}${p.refundDueCents ? `, ${p.refunded ? "refunded" : "refund due"} ${money(p.refundDueCents)}` : ""}`
        : p.expired ? `Expired ${shortDate(p.expiresAt)}` : `${p.remaining} of ${p.creditsTotal} left, use by ${shortDate(p.expiresAt)}`;

        <span class="seat-chips">${p.status === "active" ? pips(p.creditsTotal, p.creditsUsed) : ""}<span class="chip${p.status === "cancelled" || p.expired ? "" : " ok"}">${esc(state)}</span></span>

          <br>${remindBtn(b)}</p>`).join("")));
    }
    const owed = stuckRefunds();
    if (owed.length) {
      parts.push(panel("Session refunds", owed.map((b) =>
        `<p>${nameTag(b.name, b.pronouns, b.customerId)}, ${esc(dayLabel(b.start))}: <strong>${money(b.paid)}</strong> &middot; ${b.refundOwed === "auto"
          ? `the automatic refund hasn't gone through${b.refundError ? ` (Stripe said: ${esc(b.refundError)})` : ""}. It keeps retrying.`
          : "you cancelled this one. Refund it if they'd like a refund, or leave it if they're rebooking."}
          <br>${btn("Refund now", `data-refund="${esc(b.id)}"`, "Refund this session in full")} ${b.stripeUrl ? link("Payment", b.stripeUrl, "Open the payment in Stripe") : ""}</p>`).join("")));
    }
    if (zoomLeft.length) {
      parts.push(panel("Zoom meetings not deleted", zoomLeft.map((s) =>
        `<p>${nameTag(s.name, s.pronouns)}, ${esc(dayLabel(s.start))} at ${esc(clock(s.start))} is cancelled, but its Zoom meeting couldn't be deleted. It keeps retrying.${s.error ? `<br><span class="mono-hint">${esc(s.error)}</span>` : ""}</p>`).join("")
        + '<p class="hint">If the error mentions scopes or permissions, the Zoom app needs the "Delete a meeting" permission.</p>'));
    }
    if (leftOnCalendar.length) {
      parts.push(panel("Cancelled but still on your calendar", leftOnCalendar.map((s) =>
        `<p>${nameTag(s.name, s.pronouns)}, ${esc(dayLabel(s.start))} at ${esc(clock(s.start))}. It isn't happening. Removal keeps retrying; you can also delete it from your Coaching calendar by hand.${s.error ? `<br><span class="mono-hint">${esc(s.error)}</span>` : ""}</p>`).join("")));
    }
    const due = refundsDue();
    if (due.length) {
      parts.push(panel("Bundle refunds to issue", due.map((p) =>
        `<p>${nameTag(p.name, p.pronouns, p.customerId)}: <strong>${money(p.refundDueCents)}</strong> ${p.stripeUrl ? `&middot; ${link("Refund in Stripe", p.stripeUrl, "Open the payment in Stripe to refund it")}` : ""}${p.refundError ? `<br><span class="mono-hint">Stripe said: ${esc(p.refundError)}</span>` : ""}<br><span class="hint">Marked done automatically once Stripe records the refund.</span></p>`).join("")));
    }
    if (stuck.length) {
      parts.push(panel("Bookings missing a step", stuck.map((s) =>
        `<p>${nameTag(s.name, s.pronouns)}, ${esc(dayLabel(s.start))} at ${esc(clock(s.start))}: ${[!s.inCalendar ? "not in your Coaching calendar" : "", !s.emailed ? "confirmation email not sent" : ""].filter(Boolean).join(" and ")}. It retries every few minutes.</p>`).join("")));
    }
    if (failedEmails.length) {
      parts.push(panel("Emails that failed (last 14 days)", failedEmails.map((e) =>
        `<p><strong>${esc(e.kind.replace(/_/g, " "))}</strong> &middot; ${esc(fmt(e.at, { month: "short", day: "numeric", hour: "numeric", minute: "2-digit" }))}<br><span class="mono-hint">${esc(e.error || "")}</span></p>`).join("")));
    }
    parts.push(panel("Checkouts in progress", `<p>${activeHolds} time${activeHolds === 1 ? "" : "s"} currently held while someone pays.</p>`));
    return (problemsCount() ? "" : '<p class="empty">All clear. Nothing needs you right now.</p>') + parts.join("");
  }

  async function saveSettings(form) {
    const fd = new FormData(form);
    const body = {
      dayStartHour: Number(fd.get("dayStartHour")),
      dayEndHour: Number(fd.get("dayEndHour")),
      workDays: fd.getAll("workDays").map(Number),
      bufferMinutes: Number(fd.get("bufferMinutes")),
      minNoticeHours: Number(fd.get("minNoticeHours")),
      slotStepMinutes: Number(fd.get("slotStepMinutes")),
      packageValidDays: Number(fd.get("packageValidDays")),
      busyCalendars: settings.calendars ? fd.getAll("busy") : settings.values.busyCalendars,
      bookingCalendar: settings.calendars ? fd.get("bookingCalendar") : settings.values.bookingCalendar,
      remindersEnabled: fd.get("remindersEnabled") === "on",
    };
    const b = form.querySelector("[type=submit]");
    b.disabled = true;
    try {
      settings = await api("POST", "/api/admin/settings", body);
      toast("Saved.");
      render();
    } catch (err) { b.disabled = false; toast(err.message, true); }
  }

      sessions: renderSessions, students: renderStudents, bundles: renderBundles,
      attention: renderAttention, settings: renderSettings,
    }[tab]?.() ?? renderSessions();
    window.scrollTo(0, scroll);
    const search = document.getElementById("student-search");
    if (search) {
      search.addEventListener("input", () => {
        studentFilter = search.value;

  // A form dialog: submit runs `onGo` with the form; errors show as a toast.
  function formDialog(html, onGo, cls = "") {
    showDialog(`<form id="dlg-form" novalidate>${html}</form>`, cls);
    const form = $dlg.querySelector("#dlg-form");
    form.addEventListener("submit", async (e) => {
      e.preventDefault();
      if (!form.reportValidity()) return;
      const go = form.querySelector("[type=submit]");
      go.disabled = true;
      try { await onGo(form); } catch (err) { go.disabled = false; toast(err.message, true); }
    });
    return form;
  }

  const HOURS = Array.from({ length: 96 }, (_, i) => `${String(Math.floor(i / 4)).padStart(2, "0")}:${String((i % 4) * 15).padStart(2, "0")}`);
  const hhmmLabel = (hm) => { const [h, m] = hm.split(":").map(Number); return `${h % 12 || 12}:${String(m).padStart(2, "0")} ${h < 12 ? "AM" : "PM"}`; };
  const whenFields = (date, time) => `
      <label>Date<input type="date" name="date" value="${esc(date)}" required></label>
      <label>Start (Eastern)<select name="time">${HOURS.map((h) => `<option value="${h}"${h === time ? " selected" : ""}>${hhmmLabel(h)}</option>`).join("")}</select></label>`;

  // Warnings for a time: calendar clashes, outside hours, short notice; overlaps block.
  async function checkTime(form, extra) {
    const out = form.querySelector(".warnings");
    try {
      const r = await api("POST", "/api/admin/check-time", { serviceId: form.serviceId?.value || extra.serviceId, date: form.date.value, time: form.time.value, ...extra });
      const w = [];
      if (r.overlapsBooking) w.push("<p class=\"bad\">This overlaps another booking, so it can't be booked.</p>");
      if (r.calendarClash && r.calendarClash.length) w.push(`<p>Something on your calendar is at ${r.calendarClash.map((c) => `${esc(clock(c.start))} to ${esc(clock(c.end))}`).join(", ")} (within your buffer). You can still book it.</p>`);
      if (r.outsideHours) w.push("<p>This is outside your usual working hours.</p>");
      if (r.withinNotice) w.push("<p>This is sooner than your usual notice period.</p>");
      out.innerHTML = w.join("");
    } catch (err) { out.innerHTML = `<p class="bad">${esc(err.message)}</p>`; }
  }
